"""
AssetFusion360 Capstone
Step 7.6 - Gold Asset-Code Bridge

Source: Gold dim_asset
Target: Gold bridge_asset_code

Grain: One row per distinct asset code.

Preserves all candidate asset identities.
Never automatically resolves ambiguous codes.
Does not modify existing Bronze, Silver, or Gold data.
"""

import gc
import json
from collections import defaultdict
from datetime import datetime, timezone
from pathlib import Path

import pyarrow as pa
import pyarrow.parquet as pq

from kedro.io import DataCatalog, MemoryDataset
from kedro.pipeline import node, pipeline
from kedro.runner import SequentialRunner


ROOT = Path(__file__).resolve().parents[3]

SOURCE = (
    ROOT / "data" / "04_gold_analytics"
    / "dim_asset" / "part-00000.parquet"
)

GOLD_ROOT = ROOT / "data" / "04_gold_analytics"

FINAL_DIR = GOLD_ROOT / "bridge_asset_code"
STAGING_DIR = GOLD_ROOT / "_staging_bridge_asset_code"

FINAL_FILE = FINAL_DIR / "part-00000.parquet"
STAGING_FILE = STAGING_DIR / "part-00000.parquet"

MANIFEST = (
    ROOT / "reports" / "data_profiling_reports"
    / "gold_bridge_asset_code_manifest.json"
)

EXPECTED_ASSET_ROWS = 148_522
EXPECTED_BRIDGE_ROWS = 148_521
BATCH_SIZE = 5000


GOLD_SCHEMA = pa.schema([
    pa.field(
        "asset_code", pa.string(), nullable=False
    ),
    pa.field(
        "asset_match_count", pa.int32(), nullable=False
    ),
    pa.field(
        "resolved_asset_identity", pa.int64()
    ),
    pa.field(
        "candidate_asset_identities",
        pa.list_(pa.int64())
    ),
    pa.field(
        "resolution_status",
        pa.string(), nullable=False
    ),
    pa.field(
        "is_ambiguous", pa.bool_(), nullable=False
    ),
    pa.field(
        "is_placeholder", pa.bool_(), nullable=False
    ),
])


def check_outputs():
    if not SOURCE.is_file():
        raise FileNotFoundError(
            f"Gold Asset Dimension missing: {SOURCE}"
        )

    for path in [FINAL_DIR, STAGING_DIR, MANIFEST]:
        if path.exists():
            raise FileExistsError(
                f"Existing artifact must be preserved: {path}"
            )

    print("Output safety: PASS")


def inspect_source():
    with pq.ParquetFile(str(SOURCE)) as reader:
        schema = reader.schema_arrow
        rows = reader.metadata.num_rows

    assert rows == EXPECTED_ASSET_ROWS, (
        f"Unexpected asset count: {rows}"
    )

    required = {
        "asset_identity": pa.int64(),
        "asset_code": pa.string(),
        "asset_code_match_count": pa.int32(),
        "asset_code_ambiguous": pa.bool_(),
        "asset_code_placeholder": pa.bool_(),
    }

    for name, expected_type in required.items():
        assert name in schema.names, (
            f"Missing Gold dimension column: {name}"
        )

        assert schema.field(name).type.equals(
            expected_type
        ), f"Unexpected type: {name}"

    print("Source dim_asset records:", f"{rows:,}")
    print("Source schema: PASS")


def build_gold_asset_bridge():
    print("=" * 65)
    print("STEP 7.6 - BUILD GOLD ASSET-CODE BRIDGE")
    print("=" * 65)

    check_outputs()
    inspect_source()

    code_to_assets = defaultdict(list)
    placeholder_by_code = {}
    source_match_counts = {}
    source_ambiguities = {}
    identities = set()
    total_assets = 0

    with pq.ParquetFile(str(SOURCE)) as reader:
        for batch in reader.iter_batches(
            batch_size=BATCH_SIZE,
            columns=[
                "asset_identity",
                "asset_code",
                "asset_code_match_count",
                "asset_code_ambiguous",
                "asset_code_placeholder",
            ],
        ):
            for row in batch.to_pylist():
                total_assets += 1

                identity = row["asset_identity"]
                code = row["asset_code"]

                assert identity is not None
                assert identity not in identities
                assert code is not None

                identities.add(identity)
                code_to_assets[code].append(identity)

                placeholder_by_code[code] = (
                    placeholder_by_code.get(code, False)
                    or row["asset_code_placeholder"]
                )

                if code in source_match_counts:
                    assert (
                        source_match_counts[code]
                        == row["asset_code_match_count"]
                    )
                    assert (
                        source_ambiguities[code]
                        == row["asset_code_ambiguous"]
                    )
                else:
                    source_match_counts[code] = (
                        row["asset_code_match_count"]
                    )
                    source_ambiguities[code] = (
                        row["asset_code_ambiguous"]
                    )

    assert total_assets == EXPECTED_ASSET_ROWS
    assert len(identities) == EXPECTED_ASSET_ROWS
    assert len(code_to_assets) == EXPECTED_BRIDGE_ROWS

    bridge_rows = []

    for code in sorted(code_to_assets):
        candidates = sorted(code_to_assets[code])
        count = len(candidates)

        assert count == source_match_counts[code]
        assert (count > 1) == source_ambiguities[code]

        is_ambiguous = count > 1

        bridge_rows.append({
            "asset_code": code,
            "asset_match_count": count,
            "resolved_asset_identity": (
                candidates[0] if count == 1 else None
            ),
            "candidate_asset_identities": candidates,
            "resolution_status": (
                "AMBIGUOUS" if is_ambiguous else "UNIQUE"
            ),
            "is_ambiguous": is_ambiguous,
            "is_placeholder": bool(
                placeholder_by_code[code]
            ),
        })

    ambiguous = [
        row for row in bridge_rows
        if row["is_ambiguous"]
    ]

    assert len(ambiguous) == 1
    assert ambiguous[0]["asset_code"] == "B223PA"
    assert ambiguous[0]["asset_match_count"] == 2
    assert ambiguous[0]["resolved_asset_identity"] is None
    assert (
        ambiguous[0]["candidate_asset_identities"]
        == [5514, 6133]
    )

    assert sum(
        row["asset_match_count"] for row in bridge_rows
    ) == EXPECTED_ASSET_ROWS

    print("Source asset records:", f"{total_assets:,}")
    print("Distinct asset codes:", f"{len(bridge_rows):,}")
    print("Ambiguous codes:", len(ambiguous))
    print("B223PA candidates: [5514, 6133]")

    # Publish only after verifying the in-memory bridge.
    STAGING_DIR.mkdir(parents=True, exist_ok=False)

    table = pa.Table.from_pylist(
        bridge_rows,
        schema=GOLD_SCHEMA,
    )

    pq.write_table(
        table,
        str(STAGING_FILE),
        compression="snappy",
    )

    del table
    gc.collect()

    with pq.ParquetFile(str(STAGING_FILE)) as reader:
        assert (
            reader.metadata.num_rows
            == EXPECTED_BRIDGE_ROWS
        )
        assert reader.schema_arrow.equals(
            GOLD_SCHEMA,
            check_metadata=False,
        )

        verified_rows = 0
        seen_codes = set()
        ambiguous_count = 0

        for batch in reader.iter_batches(
            batch_size=BATCH_SIZE
        ):
            for row in batch.to_pylist():
                verified_rows += 1
                code = row["asset_code"]

                assert code not in seen_codes
                seen_codes.add(code)

                candidates = row[
                    "candidate_asset_identities"
                ]

                assert len(candidates) == (
                    row["asset_match_count"]
                )

                if row["is_ambiguous"]:
                    ambiguous_count += 1
                    assert (
                        row["resolved_asset_identity"]
                        is None
                    )
                else:
                    assert (
                        row["resolved_asset_identity"]
                        == candidates[0]
                    )

        assert verified_rows == EXPECTED_BRIDGE_ROWS
        assert ambiguous_count == 1

    gc.collect()

    if FINAL_DIR.exists():
        raise FileExistsError(
            "Final bridge directory appeared during build"
        )

    STAGING_DIR.rename(FINAL_DIR)

    assert FINAL_FILE.is_file()

    manifest_data = {
        "project": "AssetFusion360",
        "layer": "Gold",
        "table": "bridge_asset_code",
        "status": "COMPLETED",
        "source": SOURCE.relative_to(ROOT).as_posix(),
        "output": FINAL_FILE.relative_to(ROOT).as_posix(),
        "rows": EXPECTED_BRIDGE_ROWS,
        "columns": len(GOLD_SCHEMA),
        "source_asset_rows": EXPECTED_ASSET_ROWS,
        "unique_codes": EXPECTED_BRIDGE_ROWS - 1,
        "ambiguous_codes": 1,
        "ambiguous_code": "B223PA",
        "candidate_asset_identities": [5514, 6133],
        "created_utc": datetime.now(
            timezone.utc
        ).isoformat(),
    }

    MANIFEST.parent.mkdir(
        parents=True, exist_ok=True
    )

    with MANIFEST.open(
        "x", encoding="utf-8"
    ) as handle:
        json.dump(
            manifest_data,
            handle,
            indent=2,
        )

    print("\nGOLD BRIDGE SUMMARY")
    print("Bridge rows:", f"{verified_rows:,}")
    print("Gold columns:", len(GOLD_SCHEMA))
    print("Ambiguous codes:", ambiguous_count)
    print("Unique asset codes:", f"{verified_rows - 1:,}")
    print("Validation: PASS")
    print("GOLD ASSET BRIDGE BUILD: SUCCESS")

    return manifest_data


def create_gold_asset_bridge_pipeline():
    return pipeline([
        node(
            func=build_gold_asset_bridge,
            inputs=None,
            outputs="gold_asset_bridge_manifest",
            name="build_gold_asset_bridge",
            tags=["gold", "assets", "bridge"],
        )
    ])


def run_kedro_pipeline():
    catalog = DataCatalog({
        "gold_asset_bridge_manifest": MemoryDataset()
    })

    runner = SequentialRunner()

    runner.run(
        create_gold_asset_bridge_pipeline(),
        catalog,
    )

    result = catalog.load(
        "gold_asset_bridge_manifest"
    )

    assert result["status"] == "COMPLETED"

    print("KEDRO GOLD ASSET BRIDGE PIPELINE: SUCCESS")


if __name__ == "__main__":
    run_kedro_pipeline()

"""

from pathlib import Path

import pyarrow as pa

from kedro.io import DataCatalog, MemoryDataset

FINAL = GOLD / "dim_date"
STAGING = GOLD / "_staging_dim_date"

MANIFEST = (
    ROOT / "reports" / "data_profiling_reports"
    / "dim_date_manifest.json"

SOURCES = {
    "fact_workorders": ("workorder_date", 447, 2_231_535),
    "fact_labor": ("labor_date", 647, 3_230_423),
    "fact_transactions": ("transaction_date", 152, 755_122),
    "fact_transaction_details": ("detail_date", 461, 2_301_927),
}

SCHEMA = pa.schema([
    pa.field("date_key", pa.int32(), nullable=False),
    pa.field("calendar_date", pa.date32(), nullable=False),
    pa.field("year", pa.int16(), nullable=False),
    pa.field("quarter", pa.int8(), nullable=False),
    pa.field("month", pa.int8(), nullable=False),
    pa.field("month_name", pa.string(), nullable=False),
    pa.field("year_month", pa.string(), nullable=False),
    pa.field("day_of_month", pa.int8(), nullable=False),
    pa.field("day_of_week", pa.int8(), nullable=False),
    pa.field("day_name", pa.string(), nullable=False),
    pa.field("iso_week", pa.int8(), nullable=False),
    pa.field("is_weekend", pa.bool_(), nullable=False),
])


def check_outputs():
    for path in (FINAL, STAGING, MANIFEST):
        if path.exists():
            raise FileExistsError(
                f"Existing output preserved: {path}"
            )

    print("Output safety: PASS")


def collect_date_coverage():
    observed = set()
    source_stats = {}

    for table, (column, expected_files, expected_rows) in (
        SOURCES.items()
    ):
        files = sorted((GOLD / table).glob("*.parquet"))

        assert len(files) == expected_files, (
            f"Unexpected file count: {table}"
        )

        reference = pq.read_schema(files[0])
        assert column in reference.names

        assert pa.types.is_timestamp(
            reference.field(column).type
        ), f"Expected timestamp: {table}.{column}"

        rows = 0
        missing = 0

        for path in files:
            assert pq.read_schema(path).equals(
                reference, check_metadata=False
            )

            with pq.ParquetFile(str(path)) as reader:
                for batch in reader.iter_batches(
                    batch_size=50000,
                    columns=[column],
                ):
                    rows += batch.num_rows

                    dates = pc.cast(
                        batch.column(0),
                        pa.date32(),
                    )

                    missing += dates.null_count

                    observed.update(
                        value
                        for value in pc.unique(dates).to_pylist()
                        if value is not None
                    )

        assert rows == expected_rows

        source_stats[table] = {
            "rows": rows,
            "missing_dates": missing,
        }

        print(
            f"{table}: {rows:,} rows, "
            f"{missing:,} missing dates",
            flush=True,
        )

    assert observed, "No usable source dates"

    return observed, source_stats


def make_records(first, last):
    records = []
    current = first

    while current <= last:
        iso_weekday = current.isoweekday()

        records.append({
            "date_key": int(current.strftime("%Y%m%d")),
            "calendar_date": current,
            "year": current.year,
            "quarter": (current.month - 1) // 3 + 1,
            "month": current.month,
            "month_name": current.strftime("%B"),
            "year_month": current.strftime("%Y-%m"),
            "day_of_month": current.day,
            "day_of_week": iso_weekday,
            "day_name": current.strftime("%A"),
            "iso_week": current.isocalendar().week,
            "is_weekend": iso_weekday >= 6,
        })

        current += timedelta(days=1)

    return records


def build_dim_date():
    print("=" * 68)
    print("STEP 8.2 - BUILD DIM_DATE")
    print("=" * 68)

    check_outputs()

    observed, source_stats = collect_date_coverage()

    first = min(observed)
    last = max(observed)

    records = make_records(first, last)

    assert len(records) == (last - first).days + 1
    assert len(records) >= len(observed)

    date_keys = {r["date_key"] for r in records}
    calendar_dates = {r["calendar_date"] for r in records}

    assert len(date_keys) == len(records)
    assert len(calendar_dates) == len(records)
    assert observed.issubset(calendar_dates)

    print("\nCALENDAR VALIDATION")
    print("First date:", first)
    print("Last date:", last)
    print("Calendar rows:", len(records))
    print("Observed dates covered:", len(observed))
    print("Unique date keys: PASS")
    print("Continuous calendar: PASS")

    STAGING.mkdir(parents=True, exist_ok=False)
    staged_file = STAGING / "part-00000.parquet"

    table = pa.Table.from_pylist(
        records,
        schema=SCHEMA,
    )

    pq.write_table(
        table,
        staged_file,
        compression="snappy",
    )

    with pq.ParquetFile(str(staged_file)) as reader:
        assert reader.metadata.num_rows == len(records)
        assert reader.schema_arrow.equals(
            SCHEMA,
            check_metadata=False,
        )

        staged = reader.read().to_pylist()

    assert staged == records

    print("\nSTAGED DIM_DATE VALIDATION: PASS")

    if FINAL.exists():
        raise FileExistsError(FINAL)

    STAGING.rename(FINAL)

    manifest = {
        "project": "AssetFusion360",
        "layer": "Gold",
        "table": "dim_date",
        "status": "COMPLETED",
        "rows": len(records),
        "columns": len(SCHEMA),
        "parquet_files": 1,
        "first_date": first.isoformat(),
        "last_date": last.isoformat(),
        "observed_dates": len(observed),
        "source_stats": source_stats,
        "created_utc": datetime.now(
            timezone.utc
        ).isoformat(),
    }

    MANIFEST.parent.mkdir(
        parents=True,
        exist_ok=True,
    )

    with MANIFEST.open("x", encoding="utf-8") as handle:
        json.dump(manifest, handle, indent=2)

    print("\nGOLD DIM_DATE: SUCCESS")
    return manifest


def create_dim_date_pipeline():
    return pipeline([
        node(
            func=build_dim_date,
            inputs=None,
            outputs="dim_date_manifest",
            name="build_dim_date",
            tags=["gold", "dimension", "date"],
        )
    ])


def run_kedro_pipeline():
    catalog = DataCatalog({
        "dim_date_manifest": MemoryDataset()
    })

    SequentialRunner().run(

    )

    result = catalog.load("dim_date_manifest")
    assert result["status"] == "COMPLETED"

    print("KEDRO DIM_DATE PIPELINE: SUCCESS")


if __name__ == "__main__":
    run_kedro_pipeline()

"""
AssetFusion360 - Detail Type Dimension

Source: Gold fact_transaction_details
Output: Gold dim_detail_type

Preserves original detail type values.
Creates an explicit missing-value member if needed.
"""

from collections import Counter
from datetime import datetime, timezone
from pathlib import Path
import json

import pyarrow as pa
import pyarrow.parquet as pq


ROOT = Path(__file__).resolve().parents[3]
GOLD = ROOT / "data" / "04_gold_analytics"

SOURCE = GOLD / "fact_transaction_details"
FINAL = GOLD / "dim_detail_type"
STAGING = GOLD / "_staging_dim_detail_type"

    ROOT / "reports" / "data_profiling_reports"
    / "dim_detail_type_manifest.json"
)

EXPECTED_ROWS = 2_301_927
EXPECTED_FILES = 461

MISSING_LABEL = "<MISSING_DETAIL_TYPE>"

SCHEMA = pa.schema([
    pa.field("detail_type_key", pa.int32(), nullable=False),
    pa.field("detail_type", pa.string(), nullable=False),
    pa.field("is_missing_type", pa.bool_(), nullable=False),
    pa.field("source_detail_count", pa.int64(), nullable=False),
])


def build_dimension():
    print("=" * 68)
    print("STEP 8.2H - BUILD DIM_DETAIL_TYPE")
    print("=" * 68)

    for path in [FINAL, STAGING, MANIFEST]:
        if path.exists():
            raise FileExistsError(
                f"Existing output preserved: {path}"
            )

    files = sorted(SOURCE.glob("*.parquet"))
    assert len(files) == EXPECTED_FILES

    counts = Counter()

    for index, path in enumerate(files, 1):
        with pq.ParquetFile(str(path)) as reader:
            for batch in reader.iter_batches(
                batch_size=10000,
                columns=["detail_type"],
            ):
                counts.update(batch.column(0).to_pylist())

        if index % 100 == 0 or index == len(files):
            print(
                f"Processed files: {index}/{len(files)}",
                flush=True,
            )

    assert sum(counts.values()) == EXPECTED_ROWS

    non_null = [
        value for value in counts
        if value is not None
    ]

    assert all(isinstance(v, str) for v in non_null)
    assert MISSING_LABEL not in non_null

    records = []

    if counts[None] > 0:
        records.append({
            "detail_type_key": 0,
            "detail_type": MISSING_LABEL,
            "is_missing_type": True,
            "source_detail_count": counts[None],
        })

    for key, detail_type in enumerate(sorted(non_null), 1):
        records.append({
            "detail_type_key": key,
            "detail_type": detail_type,
            "is_missing_type": False,
            "source_detail_count": counts[detail_type],
        })

    assert records, "No detail types found"

    assert sum(
        row["source_detail_count"]
        for row in records
    ) == EXPECTED_ROWS

    assert len({
        row["detail_type_key"] for row in records
    }) == len(records)

    assert len({
        row["detail_type"] for row in records
    }) == len(records)

    print("\nDETAIL TYPE DISTRIBUTION")

    for record in records:
        print(
            f"{record['detail_type']}: "
            f"{record['source_detail_count']:,}"
        )

    STAGING.mkdir(parents=True, exist_ok=False)

    staged_file = STAGING / "part-00000.parquet"

    table = pa.Table.from_pylist(
        records,
        schema=SCHEMA,
    )

    pq.write_table(
        table,
        staged_file,
        compression="snappy",
    )

    with pq.ParquetFile(str(staged_file)) as reader:
        assert reader.metadata.num_rows == len(records)

        assert reader.schema_arrow.equals(
            SCHEMA,
            check_metadata=False
        )

        verified = reader.read().to_pylist()

    assert verified == records

    print("\nSTAGED DIMENSION VALIDATION: PASS")

    if FINAL.exists():
        raise FileExistsError(FINAL)

    STAGING.rename(FINAL)

    manifest = {
        "project": "AssetFusion360",
        "layer": "Gold",
        "table": "dim_detail_type",
        "status": "COMPLETED",
        "rows": len(records),
        "columns": len(SCHEMA),
        "parquet_files": 1,
        "source_detail_rows": EXPECTED_ROWS,
        "missing_detail_types": counts[None],
        "created_utc": datetime.now(
            timezone.utc
        ).isoformat(),

    MANIFEST.parent.mkdir(
        parents=True,
        exist_ok=True,
    )

    with MANIFEST.open("x", encoding="utf-8") as handle:
        json.dump(manifest, handle, indent=2)

    print("\nDIM_DETAIL_TYPE")
    print("Dimension rows:", len(records))
    print("Missing detail types:", counts[None])
    print("Unique dimension keys: PASS")
    print("Source-to-dimension reconciliation: PASS")
    print("Published dimension: PASS")

    print("\nSTEP 8.2H DIM_DETAIL_TYPE: PASS")

    missing_pairs = 0

    for index, path in enumerate(files, 1):
        assert pq.read_schema(path).equals(
            reference_schema,

            for batch in reader.iter_batches(
                batch_size=10000,
                columns=FIELDS,
            ):
                for row in batch.to_pylist():
                    total_rows += 1

                    code = row["part_code"]
                    identity = row["part_identity"]

                    if code is None:
                        missing_codes += 1

                    if identity is None:
                        missing_identities += 1

                    if code is None and identity is None:
                        missing_pairs += 1

                    # A complete part relationship requires
                    # both code and source identity.
                    if code is None or identity is None:
                        continue

                    attrs = (
                        identity,
                        row["part_description"],
                        row["part_class"],
                        row["part_uom"],
                    )

                    if code in attributes:
                        assert attributes[code] == attrs, (
                            f"Conflicting attributes: {code}"
                        )
                    else:
                        attributes[code] = attrs

                    if identity in identity_to_code:
                        assert identity_to_code[identity] == code, (
                            f"Identity/code conflict: {identity}"
                        )
                    else:
                        identity_to_code[identity] = code

                    counts[code] += 1

        if index % 100 == 0 or index == len(files):
            print(
                f"Processed files: {index}/{len(files)}",
                flush=True,
            )

    assert total_rows == EXPECTED_ROWS

    assert missing_identities == EXPECTED_MISSING

def inspect_sources():
    files = sorted(SILVER_DIR.glob("*.parquet"))

    assert len(files) == EXPECTED_FILES, (
        f"Unexpected Labor file count: {len(files)}"
    )

    reference_schema = pq.read_schema(files[0])
    total_rows = 0

    for path in files:
        current_schema = pq.read_schema(path)

        assert current_schema.equals(
            reference_schema,
            check_metadata=False,
        ), f"Schema mismatch: {path.name}"

        total_rows += pq.read_metadata(path).num_rows

    assert total_rows == EXPECTED_ROWS
    assert len(reference_schema) == 38

    for gold_name, silver_name in SOURCE_MAPPING.items():
        assert silver_name in reference_schema.names, (
            f"Missing Silver field: {silver_name}"
        )

        source_type = reference_schema.field(
            silver_name
        ).type

        gold_type = GOLD_SCHEMA.field(
            gold_name
        ).type

        assert source_type.equals(gold_type), (
            f"Type mismatch: {silver_name}: "
            f"{source_type} != {gold_type}"
        )

    print("Silver Labor files:", len(files))
    print("Silver Labor rows:", f"{total_rows:,}")
    print("Silver Labor columns:", len(reference_schema))
    print("Gold Labor columns:", len(GOLD_SCHEMA))
    print("Labor source schema: PASS")

    return files

def load_workorder_lookup():
    files = sorted(WORKORDER_DIR.glob("*.parquet"))

    assert len(files) == 447

    lookup = {}
    total_rows = 0

    for path in files:
        with pq.ParquetFile(str(path)) as reader:
            for batch in reader.iter_batches(
                batch_size=BATCH_SIZE,
                columns=[
                    "workorder_code",
                    "workorder_identity",
                ],
            ):
                for row in batch.to_pylist():
                    total_rows += 1

                    code = row["workorder_code"]
                    identity = row["workorder_identity"]

                    assert code is not None
                    assert identity is not None
                    assert code not in lookup, (
                        f"Duplicate work-order code: {code}"
                    )

                    lookup[code] = identity

    assert total_rows == EXPECTED_WORKORDERS
    assert len(lookup) == EXPECTED_WORKORDERS

    print(
        "Gold Work Order lookup:",
        f"{len(lookup):,}"
    )
    print("Work Order business key uniqueness: PASS")

    return lookup


# ============================================================
# TRANSFORM ONE SILVER LABOR BATCH
# ============================================================

def transform_batch(batch, workorder_lookup):
    source = pa.Table.from_batches([batch])

    references = source.column(
        "boo_event"
    ).to_pylist()

    resolved_ids = []
    statuses = []

    for reference in references:
        if reference is None or not str(reference).strip():
            resolved_ids.append(None)
            statuses.append("MISSING_REFERENCE")

        elif reference in workorder_lookup:
            resolved_ids.append(
                workorder_lookup[reference]
            )
            statuses.append("MATCHED")

        else:
            resolved_ids.append(None)
            statuses.append("UNMATCHED")

    derived = {
        "workorder_identity": pa.array(
            resolved_ids,
            type=pa.int64(),
        ),
        "workorder_match_status": pa.array(
            statuses,
            type=pa.string(),
        ),
    }

    columns = []

    for field in GOLD_SCHEMA:
        name = field.name

        if name in derived:
            column = derived[name]
        else:
            column = source.column(
                SOURCE_MAPPING[name]
            )

        columns.append(column)

    result = pa.Table.from_arrays(
        columns,
        schema=GOLD_SCHEMA,
    )

    assert result.num_rows == batch.num_rows

    return result

# BUILD GOLD FACT_LABOR
# ============================================================

def build_gold_fact_labor():
    print("=" * 68)
    print("STEP 7.8 - BUILD GOLD FACT_LABOR")
    print("=" * 68)

    check_outputs()

    silver_files = inspect_sources()
    workorder_lookup = load_workorder_lookup()

    # Only create staging after source validation.
    STAGING_DIR.mkdir(
        parents=True,
        exist_ok=False,
    )

    counts = Counter()

    for index, source_file in enumerate(silver_files):
        target_file = (
            STAGING_DIR / f"part-{index:05d}.parquet"
        )

        with pq.ParquetFile(str(source_file)) as reader:
            with pq.ParquetWriter(
                str(target_file),
                GOLD_SCHEMA,
                compression="snappy",
            ) as writer:
                for batch in reader.iter_batches(
                    batch_size=BATCH_SIZE,
                    columns=list(SOURCE_MAPPING.values()),
                ):
                    gold = transform_batch(
                        batch,
                        workorder_lookup,
                    )

                    writer.write_table(gold)

                    counts["rows"] += gold.num_rows

                    for status in gold.column(
                        "workorder_match_status"
                    ).to_pylist():
                        counts[status] += 1

                    for column, name in [
                        ("negative_hours_flag", "negative_hours"),
                        ("negative_cost_flag", "negative_cost"),
                        ("zero_rate_flag", "zero_rate"),
                        ("silver_dq_issue", "dq_issues"),
                    ]:
                        counts[name] += sum(
                            value is True
                            for value in gold.column(
                                column
                            ).to_pylist()
                        )

        if (
            (index + 1) % 25 == 0
            or index + 1 == EXPECTED_FILES
        ):
            print(
                f"Files: {index + 1}/{EXPECTED_FILES}"
                f" | Rows: {counts['rows']:,}",
                flush=True,
            )

    # ========================================================
    # VALIDATE BUSINESS COUNTS
    # ========================================================

    assert counts["rows"] == EXPECTED_ROWS

    assert counts["MATCHED"] == 3_217_345
    assert counts["UNMATCHED"] == 13_078
    assert counts["MISSING_REFERENCE"] == 0

    assert counts["negative_hours"] == 74_609
    assert counts["negative_cost"] == 71_374
    assert counts["zero_rate"] == 188_165

    assert counts["dq_issues"] == 0

    # ========================================================
    # VALIDATE STAGED PARQUET
    # ========================================================

    staged_files = sorted(
        STAGING_DIR.glob("*.parquet")
    )

    assert len(staged_files) == EXPECTED_FILES

    verified_rows = 0

    for path in staged_files:
        assert pq.read_schema(path).equals(
            GOLD_SCHEMA,
            check_metadata=False,
        ), f"Gold schema mismatch: {path.name}"

        verified_rows += pq.read_metadata(
            path
        ).num_rows

    assert verified_rows == EXPECTED_ROWS

    print("\nSTAGED GOLD LABOR VALIDATION")
    print("Labor rows:", f"{verified_rows:,}")
    print("Parquet files:", len(staged_files))
    print("Matched Work Orders:", f"{counts['MATCHED']:,}")
    print("Unmatched Work Orders:", f"{counts['UNMATCHED']:,}")
    print("Negative hours:", f"{counts['negative_hours']:,}")
    print("Negative costs:", f"{counts['negative_cost']:,}")
    print("Zero rates:", f"{counts['zero_rate']:,}")
    print("Staged validation: PASS")

    # ========================================================
    # PUBLISH WITHOUT OVERWRITING
    # ========================================================

    gc.collect()

    if FINAL_DIR.exists():
        raise FileExistsError(FINAL_DIR)

    STAGING_DIR.rename(FINAL_DIR)

    # ========================================================
    # WRITE COMPLETION MANIFEST
    # ========================================================

    manifest = {
        "project": "AssetFusion360",
        "layer": "Gold",
        "table": "fact_labor",
        "status": "COMPLETED",
        "source": SILVER_DIR.relative_to(ROOT).as_posix(),
        "workorder_source": (
            WORKORDER_DIR.relative_to(ROOT).as_posix()
        ),
        "output": FINAL_DIR.relative_to(ROOT).as_posix(),
        "rows": verified_rows,
        "columns": len(GOLD_SCHEMA),
        "parquet_files": len(staged_files),
        "matched_workorder_rows": counts["MATCHED"],
        "unmatched_workorder_rows": counts["UNMATCHED"],
        "missing_reference_rows": counts["MISSING_REFERENCE"],
        "negative_hours_rows": counts["negative_hours"],
        "negative_cost_rows": counts["negative_cost"],
        "zero_rate_rows": counts["zero_rate"],
        "silver_dq_issue_rows": counts["dq_issues"],
        "created_utc": datetime.now(
            timezone.utc
        ).isoformat(),
    }

    MANIFEST.parent.mkdir(
        parents=True,
        exist_ok=True,
    )

    with MANIFEST.open(
        "x",
        encoding="utf-8",
    ) as handle:
        json.dump(manifest, handle, indent=2)

    print("\nGOLD FACT_LABOR BUILD: SUCCESS")

    return manifest


# ============================================================
# KEDRO PIPELINE
# ============================================================

def create_gold_fact_labor_pipeline():
    return pipeline([
        node(
            func=build_gold_fact_labor,
            inputs=None,
            outputs="gold_fact_labor_manifest",
            name="build_gold_fact_labor",
            tags=["gold", "labor", "fact"],
        )
    ])


def run_kedro_pipeline():
    catalog = DataCatalog({
        "gold_fact_labor_manifest": MemoryDataset()
    })

    runner = SequentialRunner()

    runner.run(
        create_gold_fact_labor_pipeline(),
        catalog,
    )

    result = catalog.load("gold_fact_labor_manifest")

    assert result["status"] == "COMPLETED"

    print("KEDRO GOLD FACT_LABOR PIPELINE: SUCCESS")


if __name__ == "__main__":
    run_kedro_pipeline()

def load_lookup(directory, code_field, identity_field, count):
    files = sorted(directory.glob("*.parquet"))
    lookup = {}

    for path in files:
        with pq.ParquetFile(str(path)) as reader:
            for batch in reader.iter_batches(
                batch_size=BATCH_SIZE,

                data = batch.to_pydict()

                for code, identity in zip(
                    data[code_field],

                    assert code is not None

                    lookup[code] = identity

    assert len(lookup) == count
    return lookup

def resolve_codes(codes, lookup):
    identities = []
    statuses = []

    for code in codes:
        if code is None or not str(code).strip():
            identities.append(None)
            statuses.append("MISSING_REFERENCE")
        elif code in lookup:
            identities.append(lookup[code])
            statuses.append("MATCHED")
        else:
            identities.append(None)
            statuses.append("UNMATCHED")

    return identities, statuses


def transform_batch(batch, transactions, workorders):
    source = pa.Table.from_batches([batch])

    transaction_ids, transaction_statuses = resolve_codes(
        source.column("trl_trans").to_pylist(),
        transactions,
    )

    workorder_ids, workorder_statuses = resolve_codes(
        source.column("trl_event").to_pylist(),
        workorders,
    )

    derived = {
        "transaction_identity": pa.array(
            transaction_ids, type=pa.int64()
        ),
        "transaction_match_status": pa.array(
            transaction_statuses, type=pa.string()
        ),
        "workorder_identity": pa.array(
            workorder_ids, type=pa.int64()
        ),
        "workorder_match_status": pa.array(
            workorder_statuses, type=pa.string()
        ),
    }

    columns = []

    for field in GOLD_SCHEMA:
        if field.name in derived:
            columns.append(derived[field.name])
        else:
            columns.append(
                source.column(SOURCE_MAPPING[field.name])
            )

    result = pa.Table.from_arrays(
        columns,
        schema=GOLD_SCHEMA,
    )

    assert result.num_rows == batch.num_rows
    return result

def load_asset_bridge():
    bridge = {}

    with pq.ParquetFile(str(BRIDGE_FILE)) as reader:
        if reader.metadata.num_rows != 148_521:

        for batch in reader.iter_batches(

                    assert identity is not None
                    assert row["is_ambiguous"] is False
                else:
                    assert identity is None
                    assert row["is_ambiguous"] is True

                bridge[code] = {
                    "identity": identity,
                    "status": status,
                    "ambiguous": row["is_ambiguous"],
                    "placeholder": row["is_placeholder"],
                }

    assert len(bridge) == 148_521
    assert bridge["B223PA"]["status"] == "AMBIGUOUS"

def transform_batch(batch, bridge):
    source = pa.Table.from_batches([batch])

    asset_codes = source.column(
        "evt_object"
    ).to_pylist()

    resolved = []
    statuses = []
    ambiguous = []
    placeholders = []

    for code in asset_codes:
        if code not in bridge:
            raise ValueError(
                f"Unmatched work-order asset code: {code!r}"
            )

        match = bridge[code]

        resolved.append(match["identity"])
        statuses.append(match["status"])
        ambiguous.append(match["ambiguous"])
        placeholders.append(match["placeholder"])

    derived = {
        "resolved_asset_identity": pa.array(
            resolved, type=pa.int64()
        ),
        "asset_resolution_status": pa.array(
            statuses, type=pa.string()
        ),
        "asset_reference_ambiguous": pa.array(
            ambiguous, type=pa.bool_()
        ),
        "asset_code_placeholder": pa.array(
            placeholders, type=pa.bool_()
        ),
    }

    columns = []

    for field in GOLD_SCHEMA:
        name = field.name

        if name in derived:
            columns.append(derived[name])
        else:
            columns.append(
                source.column(SOURCE_MAPPING[name])
            )

    result = pa.Table.from_arrays(
        columns, schema=GOLD_SCHEMA
    )

    assert result.num_rows == batch.num_rows

    return result

def aggregate_labor(files):

            for batch in reader.iter_batches(
                batch_size=BATCH_SIZE,
                columns=SOURCE_FIELDS,
            ):
                data = batch.to_pydict()

                for i in range(batch.num_rows):
                    date = data["labor_date"][i]
                    status = data[
                        "workorder_match_status"
                    ][i]
                    hours = data["labor_hours"][i]
                    cost = data["labor_cost"][i]
                    rate = data["labor_rate"][i]

                    if date is None:
                        raise ValueError(
                            "Missing labor reporting date"
                        )

                    assert status in {
                        "MATCHED",
                        "UNMATCHED",
                        "MISSING_REFERENCE",
                    }

                    assert hours is not None
                    assert cost is not None
                    assert rate is not None

                    month = date.strftime("%Y-%m")
                    key = (month, status)
                    months.add(month)

                    group = groups[key]
                    group["labor_record_count"] += 1

                    signed_totals[key]["hours"] += hours
                    signed_totals[key]["cost"] += cost

                    negative_cost = cost < 0
                    zero_rate = rate == 0

                    assert (
                        data["negative_hours_flag"][i]
                        == negative_hours
                    )
                    assert (
                        data["negative_cost_flag"][i]
                        == negative_cost
                    )
                    assert (
                        data["zero_rate_flag"][i]
                        == zero_rate
                    )

                    group["negative_cost_count"] += int(
                        negative_cost
                    )
                    group["zero_rate_count"] += int(
                        zero_rate
                    )

                    counts[status] += 1
                    counts["negative_hours"] += int(
                        negative_hours
                    )
                    counts["negative_cost"] += int(
                        negative_cost
                    )

def aggregate_transactions(files):
    groups = defaultdict(Counter)
    totals = Counter()

    for index, path in enumerate(files, 1):

                data = batch.to_pydict()

                for i in range(batch.num_rows):
                    date = data["transaction_date"][i]
                    kind = data["transaction_type"][i]
                    status = data["transaction_status"][i]

                    if date is None:
                        month = "<MISSING_DATE>"

                        status = "<MISSING_STATUS>"

                    group = groups[key]

                    missing_date = int(date is None)
                    invalid_date = int(
                        data["dq_transaction_date_invalid"][i]
                        is True

                    group["invalid_date_flag_count"] += invalid_date
                    group["silver_dq_issue_count"] += dq_issue
