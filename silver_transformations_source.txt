def utc_now():
    return datetime.now(timezone.utc).isoformat()


def relative(path):
    return path.relative_to(ROOT).as_posix()


def sha256_file(path):
    digest = hashlib.sha256()

    with path.open("rb") as handle:
        for chunk in iter(
            lambda: handle.read(1024 * 1024),
            b"",
        ):
            digest.update(chunk)

    return digest.hexdigest()


def is_missing(value):
    return (
        value is None
        or str(value).strip() == ""
    )


def parse_integer(value):
    if is_missing(value):
        return None, False

    text = str(value).strip()

    if not re.fullmatch(r"[+-]?\d+", text):
        return None, True

    try:
        result = int(text)
    except ValueError:
        return None, True

    if not -(2**63) <= result < 2**63:
        return None, True

    return result, False


def parse_decimal(value):
    if is_missing(value):
        return None, False

    try:
        result = Decimal(str(value).strip())

        if not result.is_finite():
            return None, True

        result = result.quantize(
            Decimal("0.000001")
        )

        # Decimal(20,6) allows at most
        # 14 digits before the decimal point.
        if abs(result) >= Decimal("100000000000000"):
            return None, True

        return result, False

    except (InvalidOperation, ValueError, OverflowError):
        return None, True

def parse_timestamp(value):
    if is_missing(value):
        return None, False

    text = str(value).strip()

    # Source dates include repeated spaces,
    # such as "Feb 20 2009  9:36AM".

    text = re.sub(r"\s+", " ", text)

    formats = [
        "%b %d %Y %I:%M%p",
        "%b %d %Y %I:%M:%S%p",
        "%b %d %Y %H:%M:%S",
        "%Y-%m-%dT%H:%M:%S.%f",
        "%Y-%m-%dT%H:%M:%S",
        "%Y-%m-%d %H:%M:%S.%f",
        "%Y-%m-%d %H:%M:%S",
        "%Y-%m-%d",
    ]

    for date_format in formats:
        try:
            parsed = datetime.strptime(
                text,
                date_format
            )

            return parsed, False

        except ValueError:
            continue

    return None, True

def inspect_bronze():
    bronze = pq.ParquetFile(
        str(BRONZE_FILE)
    )

    schema = bronze.schema_arrow

    if bronze.metadata.num_rows != EXPECTED_ROWS:
        raise ValueError(
            "Bronze record count mismatch."
        )

    if len(schema) != EXPECTED_BRONZE_COLUMNS:
        raise ValueError(
            "Bronze column count mismatch."
        )

    for column in REQUIRED_COLUMNS:
        if column not in schema.names:
            raise ValueError(
                f"Missing required Bronze column: {column}"
            )

    for field in NEW_FIELDS:
        if field.name in schema.names:
            raise ValueError(
                f"Silver field already exists: {field.name}"
            )

    if any(
        not field.type.equals(pa.string())
        for field in schema
    ):
        raise ValueError(
            "Unexpected Bronze field type."
        )

    print("Bronze records:", bronze.metadata.num_rows)
    print("Bronze columns:", len(schema))
    print("Bronze schema: PASS")

    return bronze, schema

# 6. TRANSFORM ONE RECORD

def transform_record(record):
    result = dict(record)

    asset_code = record.get("OBJ_CODE")

    identity, identity_invalid = parse_integer(
        record.get("OBJ_SQLIDENTITY")
    )

    commissioned, commissioned_invalid = (
        parse_timestamp(
            record.get("OBJ_COMMISS")
        )
    )

    created, created_invalid = parse_timestamp(
        record.get("OBJ_CREATED")
    )

    value, value_invalid = parse_decimal(
        record.get("OBJ_VALUE")
    )

    code_missing = is_missing(asset_code)

    code_placeholder = (
        not code_missing
        and str(asset_code).strip().upper()
        in {"*", "N/A", "NA", "NULL", "UNKNOWN"}
    )

    # An absent identifier is an issue,
    # distinct from a parsing failure.
    identity_missing = is_missing(
        record.get("OBJ_SQLIDENTITY")
    )

    any_issue = any([
        code_missing,
        code_placeholder,
        identity_invalid,
        identity_missing,
        commissioned_invalid,
        created_invalid,
        value_invalid,
    ])

    result.update({
        "asset_identity_int": identity,
        "asset_commissioned_ts": commissioned,
        "asset_created_ts": created,
        "asset_value_decimal": value,
        "dq_asset_code_missing": code_missing,
        "dq_asset_code_placeholder": code_placeholder,
        "dq_asset_identity_invalid": (
            identity_invalid or identity_missing
        ),
        "dq_commission_date_invalid": (
            commissioned_invalid
        ),
        "dq_created_date_invalid": created_invalid,
        "dq_asset_value_invalid": value_invalid,
        "dq_any_issue": any_issue,
    })

    return result

# 7. STREAMING SILVER TRANSFORMATION
# ============================================================

def transform_assets():
    print("\n" + "=" * 70)
    print("ASSETFUSION360 - STEP 6.2")
    print("AMPS ASSETS SILVER TRANSFORMATION")
    print("=" * 70)

    check_safety()

    bronze, bronze_schema = inspect_bronze()

    bronze_hash_before = sha256_file(
        BRONZE_FILE
    )

    silver_schema = pa.schema(
        list(bronze_schema) + NEW_FIELDS
    )

    STAGING_DIR.mkdir(
        parents=True,
        exist_ok=False
    )

    staging_file = (
        STAGING_DIR / "part-00000.parquet"
    )

    quality_counts = {
        field.name: 0
        for field in NEW_FIELDS
        if field.name.startswith("dq_")
    }

    rows_written = 0

    with pq.ParquetWriter(
        str(staging_file),
        silver_schema,
        compression="snappy",
        version="2.6",
    ) as writer:

        for batch in bronze.iter_batches(
            batch_size=BATCH_SIZE
        ):
            input_table = pa.Table.from_batches(
                [batch]
            )

            source_rows = input_table.to_pylist()

            transformed_rows = [
                transform_record(record)
                for record in source_rows
            ]

            for record in transformed_rows:
                for flag in quality_counts:
                    if record[flag]:
                        quality_counts[flag] += 1

            silver_table = pa.Table.from_pylist(
                transformed_rows,
                schema=silver_schema,
            )

            writer.write_table(silver_table)

            rows_written += silver_table.num_rows

            print(
                f"Processed: {rows_written:,} "
                f"of {EXPECTED_ROWS:,} assets",
                flush=True,
            )

    # ========================================================
    # 8. VALIDATE STAGED SILVER PARQUET
    # ========================================================

    print("\nVALIDATING SILVER DATASET")
    print("-" * 65)

    silver_metadata = pq.read_metadata(
        staging_file
    )

    silver_actual_schema = pq.read_schema(
        staging_file
    )

    if rows_written != EXPECTED_ROWS:
        raise ValueError(
            "Silver row reconciliation failed."
        )

    if silver_metadata.num_rows != EXPECTED_ROWS:
        raise ValueError(
            "Silver Parquet row count mismatch."
        )

    if not silver_actual_schema.equals(
        silver_schema,
        check_metadata=False,
    ):
        raise ValueError(
            "Silver Parquet schema mismatch."
        )

    # Verify the Bronze input remains unchanged.
    bronze_hash_after = sha256_file(
        BRONZE_FILE
    )

    if bronze_hash_after != bronze_hash_before:
        raise RuntimeError(
            "Bronze source checksum changed."
        )

    # Validate row-count preservation and original
    # Bronze field values against the staged Silver.
    # Check batches in the same physical row order.

    bronze_verify = pq.ParquetFile(
        str(BRONZE_FILE)
    )

    silver_verify = pq.ParquetFile(
        str(staging_file)
    )

    source_columns = bronze_schema.names

    silver_original_batches = (
        silver_verify.iter_batches(
            batch_size=BATCH_SIZE,
            columns=source_columns,
        )
    )

    verified_original_rows = 0

    for bronze_batch, silver_batch in zip(
        bronze_verify.iter_batches(
            batch_size=BATCH_SIZE,
            columns=source_columns,
        ),
        silver_original_batches,
        strict=True,
    ):
        if not bronze_batch.equals(
            silver_batch,
            check_metadata=False,
        ):
            raise ValueError(
                "Original Bronze values were modified "
                "during Silver transformation."
            )

        verified_original_rows += (
            bronze_batch.num_rows
        )

    if verified_original_rows != EXPECTED_ROWS:
        raise ValueError(
            "Original-field verification count mismatch."
        )

    print("Silver records:", f"{rows_written:,}")
    print("Silver columns:", len(silver_schema))
    print("Bronze original fields preserved: PASS")
    print("Bronze SHA-256 unchanged: PASS")
    print("Silver row reconciliation: PASS")
    print("Silver schema validation: PASS")

    # ========================================================
    # 9. PUBLISH FINAL SILVER DATA
    # ========================================================

    if SILVER_DIR.exists():
        raise FileExistsError(
            "Silver output appeared during processing."
        )

    STAGING_DIR.rename(SILVER_DIR)

    manifest = {
        "project": "AssetFusion360",
        "phase": "Silver",
        "step": "6.2",
        "dataset": "AMPS Assets",
        "status": "COMPLETED",
        "engine": "PyArrow",
        "orchestration": "Kedro SequentialRunner",
        "bronze_source": relative(BRONZE_FILE),
        "silver_output": relative(SILVER_FILE),
        "bronze_sha256": bronze_hash_before,
        "bronze_rows": EXPECTED_ROWS,
        "silver_rows": rows_written,
        "bronze_columns": len(bronze_schema),
        "silver_columns": len(silver_schema),
        "original_source_fields_preserved": True,
        "row_count_preserved": True,
        "quality_flag_counts": quality_counts,
        "silver_parquet_row_groups": (
            silver_metadata.num_row_groups
        ),
        "silver_parquet_size_bytes": (
            SILVER_FILE.stat().st_size
        ),
        "transformation_rules": [
            "Preserve all original Bronze fields.",
            "Parse selected date values into timestamps.",
            "Parse asset identity into nullable int64.",
            "Parse asset value into nullable decimal(20,6).",
            "Preserve missing values.",
            "Flag malformed nonempty dates and values.",
            "Flag missing and placeholder asset codes.",
            "Never filter or deduplicate source records.",
        ],
        "created_utc": utc_now(),
    }

    MANIFEST_FILE.parent.mkdir(
        parents=True,
        exist_ok=True
    )

    with MANIFEST_FILE.open(
        "x",
        encoding="utf-8"
    ) as handle:
        json.dump(
            manifest,
            handle,
            indent=2,
            ensure_ascii=False,
        )

    print("\n" + "=" * 70)
    print("AMPS ASSETS SILVER SUMMARY")
    print("=" * 70)
    print("Bronze rows:", f"{EXPECTED_ROWS:,}")
    print("Silver rows:", f"{rows_written:,}")
    print("Original columns:", len(bronze_schema))
    print("Silver columns:", len(silver_schema))

    print("\nQUALITY FLAG COUNTS")
    for flag, count in quality_counts.items():
        print(f"{flag}: {count:,}")

    print("\nSilver output:", SILVER_FILE)
    print("Manifest:", MANIFEST_FILE)
    print("\nASSETS SILVER TRANSFORMATION: SUCCESS")

    return manifest


# ============================================================
# 10. KEDRO PIPELINE
# ============================================================

def create_silver_assets_pipeline():
    return pipeline([
        node(
            func=transform_assets,
            inputs=None,
            outputs="amps_assets_silver_manifest",
            name="transform_amps_assets_silver",
            tags=["silver", "amps", "assets"],
        )
    ])


# ============================================================
# 11. KEDRO RUNNER
# ============================================================

def run_kedro_pipeline():
    catalog = DataCatalog({
        "amps_assets_silver_manifest": (
            MemoryDataset()
        )
    })

    runner = SequentialRunner()

    runner.run(
        create_silver_assets_pipeline(),
        catalog,
    )

    manifest = catalog.load(
        "amps_assets_silver_manifest"
    )

    if (
        not isinstance(manifest, dict)
        or manifest.get("status") != "COMPLETED"
    ):
        raise RuntimeError(
            "Kedro Silver Assets manifest validation failed."
        )

    print("\nKEDRO SILVER ASSETS PIPELINE: SUCCESS")


if __name__ == "__main__":
    run_kedro_pipeline()

def parse_integer(value):

def parse_decimal(value):

        return None, False

    try:
        number = Decimal(str(value).strip())

        if not number.is_finite():
            return None, True

@lru_cache(maxsize=100000)
def parse_date_cached(text):
    normalized = re.sub(
        r"\s+",
        " ",
        text.strip()
    )

    formats = [
        "%b %d %Y %I:%M%p",
        "%b %d %Y %I:%M:%S%p",
        "%Y-%m-%dT%H:%M:%S.%f",
        "%Y-%m-%dT%H:%M:%S",
        "%Y-%m-%d %H:%M:%S",
        "%Y-%m-%d",
    ]

    for fmt in formats:
        try:
            return datetime.strptime(
                normalized,
                fmt
            )
        except ValueError:
            continue

    return None


def parse_date(value):
    if missing(value):
        return None, False

    parsed = parse_date_cached(
        str(value)
    )

    return parsed, parsed is None

def transform_row(source):
    result = dict(source)

    identity, identity_invalid = parse_integer(
        source.get("boo_sqlidentity")
    )

    event, event_invalid = parse_integer(
        source.get("boo_event")
    )

    labor_date, date_invalid = parse_date(
        source.get("boo_date")
    )

    hours, hours_invalid = parse_decimal(
        source.get("boo_hours")
    )

    rate, rate_invalid = parse_decimal(
        source.get("boo_rate")
    )

    cost, cost_invalid = parse_decimal(
        source.get("boo_cost")
    )

    result.update({
        "labor_identity_int": identity,
        "labor_workorder_id_int": event,
        "labor_date_ts": labor_date,
        "labor_hours_decimal": hours,
        "labor_rate_decimal": rate,
        "labor_cost_decimal": cost,
    })

    flags = {
        "dq_labor_identity_invalid": (
            identity_invalid
            or missing(source.get("boo_sqlidentity"))
        ),
        "dq_workorder_reference_invalid": (
            event_invalid
            or missing(source.get("boo_event"))
        ),
        "dq_labor_date_invalid": date_invalid,
        "dq_labor_hours_invalid": hours_invalid,
        "dq_labor_rate_invalid": rate_invalid,
        "dq_labor_cost_invalid": cost_invalid,
        "dq_negative_labor_hours": (
            hours is not None and hours < 0
        ),
        "dq_negative_labor_cost": (
            cost is not None and cost < 0
        ),
        "dq_zero_labor_rate": (
            rate is not None and rate == 0
        ),
    }

    # Distinguish data errors from valid financial
    # reversals and zero-rate transactions.
    error_flags = [
        "dq_labor_identity_invalid",
        "dq_workorder_reference_invalid",
        "dq_labor_date_invalid",
        "dq_labor_hours_invalid",
        "dq_labor_rate_invalid",
        "dq_labor_cost_invalid",
    ]

    flags["dq_any_issue"] = any(
        flags[name] for name in error_flags
    )

    result.update(flags)

    return result

def inspect_sources():
    files = sorted(
        BRONZE_DIR.glob("*.parquet")
    )

    if len(files) != EXPECTED_FILES:
        raise ValueError(
            f"Expected {EXPECTED_FILES} files, "
            f"found {len(files)}"
        )

    schema = pq.read_schema(files[0])

    if len(schema) != EXPECTED_BRONZE_COLUMNS:
        raise ValueError(
            "Unexpected Bronze column count"
        )

    required_columns = [
        "boo_event",
        "boo_sqlidentity",
        "boo_date",
        "boo_hours",
        "boo_rate",
        "boo_cost",
    ]

    for field in required_columns:
        if field not in schema.names:
            raise ValueError(
                f"Missing Bronze field: {field}"
            )

    for field in schema:
        if not field.type.equals(pa.string()):
            raise ValueError(
                f"Unexpected field type: {field.name}"

def missing(value):
    return value is None or str(value).strip() == ""


def parse_integer(value):
    if missing(value):
        return None, False

    text = str(value).strip()

    if not re.fullmatch(r"[+-]?\d+", text):
        return None, True

    try:
        number = int(text)
    except ValueError:
        return None, True

    if not -(2**63) <= number < 2**63:
        return None, True

    return number, False


def parse_decimal(value):
    if missing(value):
        return None, False

    try:
        number = Decimal(str(value).strip())

        if not number.is_finite():
            return None, True

        number = number.quantize(
            Decimal("0.000001")
        )

        if abs(number) >= Decimal("100000000000000"):
            return None, True

        return number, False

    except (InvalidOperation, ValueError, OverflowError):
        return None, True


@lru_cache(maxsize=100000)
def parse_date_cached(text):
    text = re.sub(r"\s+", " ", text.strip())

    formats = [
        "%b %d %Y %I:%M%p",
        "%b %d %Y %I:%M:%S%p",
        "%Y-%m-%dT%H:%M:%S.%f",
        "%Y-%m-%dT%H:%M:%S",
        "%Y-%m-%d %H:%M:%S",
        "%Y-%m-%d",
    ]

    for fmt in formats:
        try:
            return datetime.strptime(text, fmt)
        except ValueError:
            continue

    return None


def parse_date(value):
    if missing(value):
        return None, False

    result = parse_date_cached(str(value))
    return result, result is None

def transform_row(source):
    result = dict(source)
    flags = {}

    for original, target in INTEGER_FIELDS.items():
        value, invalid = parse_integer(
            source.get(original)
        )

        result[target] = value
        flags[INTEGER_FLAGS[original]] = invalid

    for original, target in DATE_FIELDS.items():
        value, invalid = parse_date(
            source.get(original)
        )

        result[target] = value
        flags[DATE_FLAGS[original]] = invalid

    for original, target in DECIMAL_FIELDS.items():
        value, invalid = parse_decimal(
            source.get(original)
        )

        result[target] = value
        flags[DECIMAL_FLAGS[original]] = invalid

    # Required identifiers must be present.
    # trl_event and par_* identifiers can be null.
    for field in [
        "trl_trans",
        "trl_sqlidentity",
    ]:
        if missing(source.get(field)):
            flags[INTEGER_FLAGS[field]] = True

    qty = result["detail_quantity_decimal"]
    price = result["detail_price_decimal"]

    flags["dq_negative_quantity"] = (
        qty is not None and qty < 0
    )
    flags["dq_negative_price"] = (
        price is not None and price < 0
    )
    flags["dq_zero_quantity"] = (
        qty is not None and qty == 0
    )
    flags["dq_zero_price"] = (
        price is not None and price == 0
    )

    # Informational financial flags do not count
    # as parsing or required-field failures.
    error_flags = (
        list(INTEGER_FLAGS.values())
        + list(DATE_FLAGS.values())
        + list(DECIMAL_FLAGS.values())
    )

    flags["dq_any_issue"] = any(
        flags[name] for name in error_flags
    )

    result.update(flags)

    return result

def transform_row(source):

def transform_transactions():
    print("=" * 70)
    print("STEP 6.5B - AMPS TRANSACTIONS SILVER")
    print("=" * 70)

    check_outputs()

    files, bronze_schema = inspect_sources()

    silver_schema = pa.schema(
        list(bronze_schema) + ADDED_FIELDS
    )

    quality_counts = Counter({
        name: 0
        for name in QUALITY_FLAGS
    })

    total_written = 0
    total_verified = 0

    STAGING_DIR.mkdir(
        parents=True,
        exist_ok=False,
    )

    for index, source_path in enumerate(files, 1):
        staged_path = STAGING_DIR / source_path.name
        written = 0

        # Read and transform. All file handles are
        # closed before the file is validated.

        with pq.ParquetFile(str(source_path)) as source:
            expected_file_rows = source.metadata.num_rows

            with pq.ParquetWriter(
                str(staged_path),
                silver_schema,
                compression="snappy",
                version="2.6",
            ) as writer:

                for batch in source.iter_batches(
                    batch_size=BATCH_SIZE
                ):
                    source_records = (
                        pa.Table.from_batches(
                            [batch]
                        ).to_pylist()
                    )

                    transformed = [
                        transform_row(record)
                        for record in source_records
                    ]

                    for record in transformed:
                        for flag in QUALITY_FLAGS:
                            if record[flag]:
                                quality_counts[flag] += 1

                    silver_table = pa.Table.from_pylist(
                        transformed,
                        schema=silver_schema,
                    )

                    writer.write_table(silver_table)

                    written += silver_table.num_rows

        if written != expected_file_rows:
            raise ValueError(
                f"Row mismatch: {source_path.name}"
            )

        # Independently verify that the Silver file
        # preserves every original Bronze value.

        verified = 0

        with pq.ParquetFile(str(source_path)) as original:
            with pq.ParquetFile(str(staged_path)) as silver:

                if silver.metadata.num_rows != written:
                    raise ValueError(
                        f"Parquet rows mismatch: {source_path.name}"
                    )

                if not silver.schema_arrow.equals(
                    silver_schema,
                    check_metadata=False,
                ):
                    raise ValueError(
                        f"Silver schema mismatch: {source_path.name}"
                    )

                original_batches = original.iter_batches(
                    batch_size=BATCH_SIZE,
                    columns=bronze_schema.names,
                )

                silver_batches = silver.iter_batches(
                    batch_size=BATCH_SIZE,
                    columns=bronze_schema.names,
                )

                for source_batch, silver_batch in zip(
                    original_batches,
                    silver_batches,
                    strict=True,
                ):
                    if not source_batch.equals(
                        silver_batch,
                        check_metadata=False,
                    ):
                        raise ValueError(
                            "Bronze values changed: "
                            f"{source_path.name}"
                        )

                    verified += source_batch.num_rows

        if verified != written:
            raise ValueError(
                f"Original value count mismatch: "
                f"{source_path.name}"
            )

        total_written += written
        total_verified += verified

        print(
            f"[{index}/{EXPECTED_FILES}] "
            f"{source_path.name}: {written:,} rows "
            f"| total {total_written:,}",
            flush=True,
        )

    # ========================================================
    # 8. FINAL STAGING RECONCILIATION
    # ========================================================

    if total_written != EXPECTED_ROWS:
        raise ValueError(
            "Global Silver row count mismatch"
        )

        raise ValueError(
            "Global Bronze preservation check failed"
        )

        STAGING_DIR.glob("*.parquet")

        raise ValueError(

    print("-" * 60)

    print("Silver records:", f"{staged_rows:,}")
    print("Silver files:", len(staged_files))
    print("Silver columns:", len(silver_schema))
    print("Original Bronze values preserved: PASS")

    # ========================================================
    # 9. WINDOWS-SAFE PUBLICATION
    # ========================================================

    del staged_files
    gc.collect()

    if SILVER_DIR.exists():
        raise FileExistsError(
            "Final Silver directory appeared during processing"
        )

    # Rename only after all readers and writers close.
    # If publication fails, preserve staging for recovery.
    STAGING_DIR.rename(SILVER_DIR)

def parse_decimal(value):
    if missing(value):
        return None, False

    try:
        result = Decimal(str(value).strip())

        if not result.is_finite():
            return None, True

        result = result.quantize(
            Decimal("0.000001")
        )

        if abs(result) >= Decimal("100000000000000"):
            return None, True

        return result, False

    except (InvalidOperation, ValueError, OverflowError):
        return None, True

    total = 0

    for path in files:
        metadata = pq.read_metadata(path)
        file_schema = pq.read_schema(path)

        if not schema.equals(
            file_schema,
            check_metadata=False,
        ):
            raise ValueError(
                f"Schema mismatch: {path.name}"
            )

        total += metadata.num_rows

    if total != EXPECTED_ROWS:
        raise ValueError(
            f"Bronze row mismatch: {total}"
        )

    print("Source files:", len(files))
    print("Source records:", f"{total:,}")
    print("Source schema: PASS")

    return files, schema


def transform_workorders():
    print("=" * 70)
    print("STEP 6.3 - AMPS WORK ORDERS SILVER")
    print("=" * 70)

    check_outputs()
    files, source_schema = inspect_sources()

    silver_schema = pa.schema(
        list(source_schema) + ADDED_FIELDS
    )

    quality_counts = Counter({
        flag: 0 for flag in ALL_FLAGS
    })

    total_written = 0
    total_verified = 0

    STAGING_DIR.mkdir(
        parents=True,
        exist_ok=False,
    )

    for file_number, source_path in enumerate(files, 1):
        staged_path = (
            STAGING_DIR / source_path.name
        )

        source_reader = pq.ParquetFile(
            str(source_path)
        )

        expected_batch_rows = (
            source_reader.metadata.num_rows
        )

        written = 0

        try:
            with pq.ParquetWriter(
                str(staged_path),
                silver_schema,
                compression="snappy",
                version="2.6",
            ) as writer:

                for batch in source_reader.iter_batches(
                    batch_size=BATCH_SIZE
                ):
                    source_table = pa.Table.from_batches(
                        [batch]
                    )

                    records = source_table.to_pylist()

                    silver_records = [
                        transform_row(row)
                        for row in records
                    ]

                    for record in silver_records:
                        for flag in ALL_FLAGS:
                            if record[flag]:
                                quality_counts[flag] += 1

                    silver_table = pa.Table.from_pylist(
                        silver_records,
                        schema=silver_schema,
                    )

                    writer.write_table(
                        silver_table
                    )

                    written += silver_table.num_rows

        finally:
            source_reader.close()

        if written != expected_batch_rows:
            raise ValueError(
                f"Row mismatch: {source_path.name}"
            )

        # Validate output on disk and verify all
        # original Bronze values are unchanged.

        source_check = pq.ParquetFile(
            str(source_path)
        )

        silver_check = pq.ParquetFile(
            str(staged_path)
        )

        try:
            if silver_check.metadata.num_rows != written:
                raise ValueError(
                    f"Parquet row mismatch: {source_path.name}"
                )

            if not silver_check.schema_arrow.equals(
                silver_schema,
                check_metadata=False,
            ):
                raise ValueError(
                    f"Silver schema mismatch: {source_path.name}"
                )

            source_batches = source_check.iter_batches(
                batch_size=BATCH_SIZE
            )

            output_batches = silver_check.iter_batches(
                batch_size=BATCH_SIZE,
                columns=source_schema.names,
            )

            file_verified = 0

            for original, output in zip(
                source_batches,
                output_batches,
                strict=True,
            ):
                if not original.equals(
                    output,
                    check_metadata=False,
                ):
                    raise ValueError(
                        "Bronze source values changed: "
                        f"{source_path.name}"
                    )

                file_verified += original.num_rows

            if file_verified != written:
                raise ValueError(
                    f"Verification count mismatch: {source_path.name}"
                )

            total_verified += file_verified

        finally:
            source_check.close()
            silver_check.close()

        total_written += written

        print(
            f"[{file_number}/{EXPECTED_FILES}] "
            f"{source_path.name}: "
            f"{written:,} rows | "
            f"total {total_written:,}",
            flush=True,
        )

    if total_written != EXPECTED_ROWS:
        raise ValueError(
            "Global Silver row count mismatch"
        )

    if total_verified != EXPECTED_ROWS:
        raise ValueError(
            "Global Bronze preservation check failed"
        )

    staged_files = list(
        STAGING_DIR.glob("*.parquet")
    )

    if len(staged_files) != EXPECTED_FILES:
        raise ValueError(
            "Staged file inventory mismatch"
        )

    print("\nFINAL STAGING VALIDATION")
    print("Silver records:", f"{total_written:,}")
    print("Silver files:", len(staged_files))
    print("Silver columns:", len(silver_schema))
    print("Original Bronze values preserved: PASS")

    # Explicitly release file handles before moving
    # the directory. Important on Windows.
    gc.collect()

    if SILVER_DIR.exists():
        raise FileExistsError(
            "Final Silver directory appeared unexpectedly"
        )

    STAGING_DIR.rename(SILVER_DIR)

    manifest = {
        "project": "AssetFusion360",
        "phase": "Silver",
        "step": "6.3",
        "dataset": "AMPS Work Orders",
        "status": "COMPLETED",
        "engine": "PyArrow",
        "orchestration": "Kedro SequentialRunner",
        "bronze_source": (
            BRONZE_DIR.relative_to(ROOT).as_posix()
        ),
        "silver_output": (
            SILVER_DIR.relative_to(ROOT).as_posix()
        ),
        "bronze_rows": EXPECTED_ROWS,
        "silver_rows": total_written,
        "bronze_columns": len(source_schema),
        "silver_columns": len(silver_schema),
        "parquet_files": EXPECTED_FILES,
        "original_source_fields_preserved": True,
        "row_count_preserved": True,
        "quality_flag_counts": dict(quality_counts),
        "created_utc": datetime.now(
            timezone.utc
        ).isoformat(),
    }

    MANIFEST_FILE.parent.mkdir(
        parents=True,
        exist_ok=True,
    )

    with MANIFEST_FILE.open(
        "x",
        encoding="utf-8",
    ) as handle:
        json.dump(
            manifest,
            handle,
            indent=2,
        )

    print("\n" + "=" * 70)
    print("WORK ORDERS SILVER SUMMARY")
    print("=" * 70)
    print("Bronze rows:", f"{EXPECTED_ROWS:,}")
    print("Silver rows:", f"{total_written:,}")
    print("Parquet files:", EXPECTED_FILES)
    print("Original columns:", len(source_schema))
    print("Silver columns:", len(silver_schema))

    print("\nQUALITY FLAGS")
    for flag, count in quality_counts.items():
        print(f"{flag}: {count:,}")

    print("\nWORK ORDERS SILVER TRANSFORMATION: SUCCESS")

    return manifest


def create_silver_workorders_pipeline():
    return pipeline([
        node(
            func=transform_workorders,
            inputs=None,
            outputs="amps_workorders_silver_manifest",
            name="transform_amps_workorders_silver",
            tags=["silver", "amps", "workorders"],
        )
    ])


def run_kedro_pipeline():
    catalog = DataCatalog({
        "amps_workorders_silver_manifest": (
            MemoryDataset()
        )
    })

    runner = SequentialRunner()

    runner.run(
        create_silver_workorders_pipeline(),
        catalog,
    )

    result = catalog.load(
        "amps_workorders_silver_manifest"
    )

    if result.get("status") != "COMPLETED":
        raise RuntimeError(
            "Silver manifest validation failed"
        )

    print("\nKEDRO WORK ORDERS SILVER PIPELINE: SUCCESS")


if __name__ == "__main__":
    run_kedro_pipeline()
